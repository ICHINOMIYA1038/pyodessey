## メモ{化|か}（トップダウン）

{一度計算|いちどけいさん}した{結果|けっか}を{覚|おぼ}えておこう。これが「メモ{化|か}」だ！

```javascript runnable
let callCount = 0;

function fibMemo(n, memo) {
  callCount++;
  if (memo === undefined) memo = {};

  // すでに計算済みならメモから返す
  if (memo[n] !== undefined) return memo[n];

  if (n <= 1) return n;

  // 計算してメモに保存
  memo[n] = fibMemo(n - 1, memo) + fibMemo(n - 2, memo);
  return memo[n];
}

callCount = 0;
let result = fibMemo(10);
console.log("fib(10) = " + result);
console.log("関数が呼ばれた回数: " + callCount + "回");

callCount = 0;
result = fibMemo(20);
console.log("fib(20) = " + result);
console.log("関数が呼ばれた回数: " + callCount + "回");

callCount = 0;
result = fibMemo(50);
console.log("fib(50) = " + result);
console.log("関数が呼ばれた回数: " + callCount + "回");
```

fib(20) が17{万回|まんかい}から39{回|かい}に{激減|げきげん}したね！

function fibTable(n) {
  if (n <= 1) return n;

  // 下から順に計算
  for (let i = 2; i <= n; i++) {
    table[i] = table[i - 1] + table[i - 2];

## {階段|かいだん}の{登|のぼ}り{方|かた}{問題|もんだい}

```javascript runnable
function climbStairs(n) {
  if (n <= 2) return n;

  let table = [];
  table[1] = 1;  // 1段: 1通り
  table[2] = 2;  // 2段: 2通り（1+1 or 2）

  for (let i = 3; i <= n; i++) {
    // i段 = (i-1)段からの1段 + (i-2)段からの2段
    table[i] = table[i - 1] + table[i - 2];
  }

  return table[n];
}

for (let n = 1; n <= 8; n++) {
  console.log(n + "段: " + climbStairs(n) + "通り");
}
```

(n-2){段|だん}から2{段登|だんのぼ}るかの2{通|とお}り。

## コインの{組|く}み{合|あ}わせ{問題|もんだい}

{次|つぎ}は「{与|あた}えられたコインで{合計|ごうけい}nを{作|つく}る{最小枚数|さいしょうまいすう}」を{求|もと}める{問題|もんだい}だ。

```javascript runnable
function minCoins(coins, amount) {
  // table[i] = 金額iを作るのに必要な最小枚数
  let table = [];
  table[0] = 0;  // 0円は0枚

  for (let i = 1; i <= amount; i++) {
    table[i] = amount + 1;  // 最初は大きな値
    for (let j = 0; j < coins.length; j++) {
      if (coins[j] <= i) {
        let prev = table[i - coins[j]];
        if (prev + 1 < table[i]) {
          table[i] = prev + 1;
        }
      }
    }
  }

  if (table[amount] > amount) return -1;
  return table[amount];
}

let coins = [1, 5, 10];
console.log("コイン: " + coins.join(", ") + "円");
console.log("11円 → 最小" + minCoins(coins, 11) + "枚");
console.log("15円 → 最小" + minCoins(coins, 15) + "枚");
console.log("23円 → 最小" + minCoins(coins, 23) + "枚");

console.log("");
let coins2 = [1, 3, 4];
console.log("コイン: " + coins2.join(", ") + "円");
console.log("6円 → 最小" + minCoins(coins2, 6) + "枚");
console.log("7円 → 最小" + minCoins(coins2, 7) + "枚");
```

## ナップサック{問題|もんだい}の{考|かんが}え{方|かた}

「{重|おも}さ{制限|せいげん}のある{袋|ふくろ}に、{最|もっと}も{価値|かち}の{高|たか}いアイテムを{詰|つ}める」{問題|もんだい}だ。

```javascript runnable
function knapsack(items, capacity) {
  let n = items.length;
  // table[i][w] = i番目までのアイテムで重さwの最大価値
  let table = [];
  for (let i = 0; i <= n; i++) {
    table[i] = [];
    for (let w = 0; w <= capacity; w++) {
      table[i][w] = 0;
    }
  }

  for (let i = 1; i <= n; i++) {
    let item = items[i - 1];
    for (let w = 1; w <= capacity; w++) {
      // アイテムを入れない場合
      table[i][w] = table[i - 1][w];
      // アイテムが入る場合、入れたほうが良いか確認
      if (item.weight <= w) {
        let withItem = table[i - 1][w - item.weight] + item.value;
        if (withItem > table[i][w]) {
          table[i][w] = withItem;
        }
      }
    }
  }

  return table[n][capacity];
}

let treasures = [
  { name: "金の指輪",   weight: 1, value: 15 },
  { name: "銀の剣",     weight: 3, value: 20 },
  { name: "魔法の杖",   weight: 4, value: 30 },
  { name: "ダイヤの盾", weight: 5, value: 25 },
];

console.log("=== 冒険者のナップサック ===");
for (let i = 0; i < treasures.length; i++) {
  let t = treasures[i];
  console.log(t.name + " (重さ:" + t.weight + " 価値:" + t.value + ")");
}

let maxWeight = 7;
console.log("\n袋の容量: " + maxWeight);
console.log("最大価値: " + knapsack(treasures, maxWeight));
```

:::sensei
ナップサック{問題|もんだい}は{冒険|ぼうけん}でまさに{使|つか}えるね！
{限|かぎ}られた{容量|ようりょう}で{最|もっと}も{価値|かち}のあるアイテムを{選|えら}ぶ。
DPなら{全|すべ}ての{組|く}み{合|あ}わせを{効率的|こうりつてき}に{調|しら}べられるんだ。
