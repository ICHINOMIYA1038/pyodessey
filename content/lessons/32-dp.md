## メモ{化|か}（トップダウン）

```python runnable
# メモ化でフィボナッチを高速化！
def fib_memo(n, memo={}):
    if n in memo:
        return memo[n]  # メモにあればそれを返す！
    if n <= 1:
        return n

    result = fib_memo(n - 1, memo) + fib_memo(n - 2, memo)
    memo[n] = result  # 結果をメモに記録
    return result

# 大きい数でもすぐ計算できる！
for i in [10, 20, 30, 40, 50]:
    print(f"fib({i}) = {fib_memo(i)}")
```

## ボトムアップ（テーブル方式）

:::sensei
もう1つの{方法|ほうほう}は「ボトムアップ」。
{小|ちい}さい{問題|もんだい}から{順番|じゅんばん}に{解|と}いて、
{大|おお}きい{問題|もんだい}の{答|こた}えを{作|つく}る{方法|ほうほう}だ。
{空|そら}の{階段|かいだん}を1{段|だん}ずつ{登|のぼ}るイメージだよ！
:::

```python runnable
# ボトムアップでフィボナッチ
def fib_table(n):
    if n <= 1:
        return n

    # テーブル（リスト）を用意
    dp = [0] * (n + 1)
    dp[0] = 0
    dp[1] = 1

    # 小さい方から順番に計算
    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]

    return dp[n]

# 結果は同じ！
for i in [10, 20, 30, 40, 50]:
    print(f"fib({i}) = {fib_table(i)}")
```

# 階段の登り方（DP）
def climb_stairs(n):
    if n <= 2:
        return n

    dp = [0] * (n + 1)
    dp[1] = 1  # 1段は1通り
    dp[2] = 2  # 2段は2通り（1+1 or 2）

    for i in range(3, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]

    return dp[n]

for n in range(1, 11):
    print(f"{n}段の階段: {climb_stairs(n)}通り")

:::sensei

:::sensei

6+5=11で2{枚|まい}...あれ、これは{正解|せいかい}だけど、

```python runnable
# コイン問題をDPで解く！
def min_coins(coins, amount):
    # dp[i] = i円を作るのに必要な最小枚数
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0  # 0円は0枚

    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i and dp[i - coin] + 1 < dp[i]:
                dp[i] = dp[i - coin] + 1

    return dp[amount]

# 1円、5円、6円のコインで11円を作る
coins = [1, 5, 6]
amount = 11

result = min_coins(coins, amount)
print(f"コイン {coins} で {amount}円 → 最小 {result}枚")
print(f"（6 + 5 = 11 で2枚！）")

# いろいろな金額で試してみよう
for a in range(1, 16):
    r = min_coins(coins, a)
    print(f"  {a}円 → {r}枚")
```

## DPテーブルの{中身|なかみ}を{見|み}てみよう

# DPテーブルがどう埋まるか見てみよう
def min_coins_verbose(coins, amount):
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    used_coin = [0] * (amount + 1)  # どのコインを使ったか

    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i and dp[i - coin] + 1 < dp[i]:
                dp[i] = dp[i - coin] + 1
                used_coin[i] = coin

    # どのコインを使ったか復元
    result_coins = []
    remaining = amount
    while remaining > 0:
        result_coins.append(used_coin[remaining])
        remaining -= used_coin[remaining]

    return dp[amount], result_coins

## トップダウンでコイン{問題|もんだい}

:::sensei
{同|おな}じ{問題|もんだい}をメモ{化|か}（トップダウン）でも{解|と}けるよ。
{好|す}きな{方|ほう}を{使|つか}えばいい！
:::

```python runnable
# メモ化版のコイン問題
def min_coins_memo(coins, amount, memo={}):
    if amount == 0:
        return 0
    if amount < 0:
        return float('inf')
    if amount in memo:
        return memo[amount]

    best = float('inf')
    for coin in coins:
        result = min_coins_memo(coins, amount - coin, memo)
        if result + 1 < best:
            best = result + 1

    memo[amount] = best
    return best

coins = [1, 5, 6]
print(min_coins_memo(coins, 11))  # 2
print(min_coins_memo(coins, 15))  # 3 (6+6+1+1+1? いや 5+5+5=15 で3枚!)
print(min_coins_memo(coins, 23))  # ?
```
